# Elasticsearch async client. Pinned to v8.x because v9.x sends an
# `Accept: ...; compatible-with=9` header that ES 7 and 8 servers reject
# with HTTP 400 media-type errors. v8.x is wire-compatible with both 7
# and 8 server versions.
elasticsearch[async]>=8.0,<9.0

# HTTP client + OpenAI-compatible vLLM client
httpx
openai

# Document extraction (Phase 14 — harvest folders / URLs / DB-column
# file references into the RAG index). Pure-python where possible so
# the wheel set stays manageable.
pypdf>=4.0          # PDF text extraction
python-docx>=1.1    # DOCX text extraction
openpyxl>=3.1       # XLSX text extraction
beautifulsoup4>=4.12  # HTML cleanup

# Cloud document sources (Phase 17.4 — Google Drive + OneDrive).
# google-api-python-client pulls google-auth in transitively but we pin
# both so the resolver is predictable. OneDrive auth + Graph calls go
# through plain httpx; we deliberately avoid ``msal`` because the
# device-code flow we use is only three endpoints and msal brings a
# big broker/cache surface area we don't want.
google-api-python-client>=2.130
google-auth>=2.30

# Audio / video transcription (Phase 17.3 — mp3/mp4/m4a/wav/webm/ogg/opus
# harvest sources). CTranslate2-backed Whisper; much faster than the
# reference openai-whisper on CPU. Requires ``ffmpeg`` on PATH for
# non-WAV decoding — install via ``apt-get install ffmpeg`` (Linux)
# or ``winget install Gyan.FFmpeg`` (Windows).
faster-whisper>=1.0.3

# IMAP email harvest (Phase 19 — mailboxes as a DocSource kind).
# Pure-python sync client; we run it inside asyncio.to_thread from
# the async harvester. ``imap_tools`` covers Gmail, Outlook, Yandex,
# Mail.ru, Zoho, FastMail, and any RFC 3501-compliant server.
imap_tools>=1.6

# OCR for scanned PDFs and image attachments (Phase 20). Tesseract
# itself is a system binary — install via the project README. The
# language packs uzb / rus / eng are required for the project's
# three target languages. ``pypdfium2`` lets us render PDF pages
# without a poppler system dependency.
pytesseract>=0.3.10
pypdfium2>=4.30
Pillow>=10.0

# Real-time file system watcher (Phase 24 — folder DocSources get
# inotify/FSEvents/ReadDirectoryChangesW push notifications so a
# new file lands in RAG within seconds instead of waiting for the
# daily Celery-beat crawl). Falls back to cron-only behaviour when
# the kernel/OS doesn't support push events.
watchdog>=4.0

# Snowflake dialect (Phase 28 — analytics warehouse adapter). The
# connector is sync; the engine wraps every call in asyncio.to_thread.
# We don't need snowflake-sqlalchemy — the bare connector covers
# what we use.
snowflake-connector-python>=3.6

# Scheduled-report cron evaluator (Phase 29). Celery ships croniter
# transitively but pinning here makes ReportSchedule's "is this due?"
# check explicit and surfaces a clear error if it gets uninstalled.
croniter>=1.4
